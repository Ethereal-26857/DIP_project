def gaussian_kernel(size=5, sigma=1.0):
    ax = np.arange(-size // 2 + 1., size // 2 + 1.)
    xx, yy = np.meshgrid(ax, ax)
    kernel = np.exp(-(xx**2 + yy**2) / (2. * sigma**2))
    return kernel / kernel.sum()

def sobel_gradients(img):
    gx = np.array([[ -1, 0, 1], [ -2, 0, 2], [ -1, 0, 1]], dtype=float)
    gy = np.array([[ -1, -2, -1], [ 0, 0, 0], [ 1, 2, 1]], dtype=float)
    grad_x = ndi.convolve(img, gx, mode='reflect')
    grad_y = ndi.convolve(img, gy, mode='reflect')
    return np.hypot(grad_x, grad_y), grad_x, grad_y

def preprocess(img):
    kernel = gaussian_kernel(5, sigma=1.0)
    return ndi.convolve(img, kernel, mode='reflect')

def brain_mask(img):
    thresh = np.percentile(img, 5) + 0.1 * (np.percentile(img, 99) - np.percentile(img, 5))
    mask = img > thresh
    labeled, num = ndi.label(mask)
    if num == 0:
        return mask
    sizes = np.bincount(labeled.ravel())
    sizes[0] = 0
    largest = np.argmax(sizes)
    brain = labeled == largest
    brain = ndi.binary_fill_holes(brain)
    return brain

def remove_brain_edge(edge, img):
    mask = brain_mask(img)
    inner = ndi.binary_erosion(mask, structure=np.ones((5,5)), iterations=20)
    return edge & inner

def postprocess(edge):
    closed = ndi.binary_closing(edge, structure=np.ones((3,3)))
    labeled, num = ndi.label(closed)
    if num == 0:
        return closed
    sizes = np.bincount(labeled.ravel())
    sizes[0] = 0
    largest = np.argmax(sizes)
    return labeled == largest

def project1_pipeline(img, threshold):
    smooth = preprocess(img)
    grad, gx, gy = sobel_gradients(smooth)
    raw_edge = grad > threshold
    inner_edge = remove_brain_edge(raw_edge, img)
    mask = postprocess(inner_edge)
    return smooth, grad, raw_edge, inner_edge, mask
解释一下这段代码

A：
这是一个基于梯度边缘检测的**脑部 MRI 颅骨剥离（skull stripping）**流水线，各函数的作用如下：

核心组件
gaussian_kernel(size=5, sigma=1.0) — 生成 5×5 的高斯平滑核，用于去噪。先创建坐标网格，用高斯公式计算权重，最后归一化使总和为 1。

sobel_gradients(img) — 用 Sobel 算子计算图像的梯度幅值和方向。gx 检测水平边缘，gy 检测垂直边缘，np.hypot 合成梯度幅值。

preprocess(img) — 对图像做高斯卷积平滑，减少噪声对边缘检测的干扰。

脑部掩膜（brain_mask）
用百分位数（第 5 和第 99 百分位）动态计算阈值，分离前景（脑组织）和背景（空气）。
ndi.label 做连通域标记，只保留最大的连通区域（即大脑），丢弃散落的小块。
binary_fill_holes 填充颅骨内部的黑洞（脑脊液等低信号区域）。
边缘处理
remove_brain_edge — 用 binary_erosion（20 次迭代）从脑掩膜向内收缩，得到"内部区域"，与边缘图做交集，只保留脑内部的边缘，去掉头皮/颅骨外轮廓。

postprocess(edge) — 先用形态学闭运算（3×3）连接断裂边缘，再做连通域筛选，只保留最大的一块，去除孤立噪点。

流水线入口
project1_pipeline(img, threshold) 串联上述步骤：

高斯平滑 → smooth
Sobel 梯度 + 阈值化 → raw_edge
用脑掩膜过滤外部边缘 → inner_edge
后处理去噪 → mask
返回值包含中间结果（smooth, grad, raw_edge 等），便于可视化和调试。

问题分析
1. 固定阈值 T=70 不适应不同图像
T1（max=769）和 T2（max=1127）的梯度幅值分布差异很大，用同一个绝对阈值不合适。

修改方案： 使用梯度百分位数作为自适应阈值，例如取梯度第 92 百分位：


def adaptive_threshold(grad, percentile=92):
    return np.percentile(grad, percentile)
然后把 project1_pipeline 的 threshold 参数改为 percentile，内部调用 T = np.percentile(grad, percentile)。

line 35-40 的 postprocess 只保留 largest 连通分量。T2 图像中 MS 病灶通常是多发性的。

def gaussian_kernel(size=5, sigma=2.0):
    ax = np.arange(-size // 2 + 1., size // 2 + 1.)
    xx, yy = np.meshgrid(ax, ax)
    kernel = np.exp(-(xx**2 + yy**2) / (2. * sigma**2))
    return kernel / kernel.sum()

def sobel_gradients(img, operator='sobel'):
    """支持 sobel / scharr 两种算子"""
    if operator == 'scharr':
        gx = np.array([[-3, 0, 3], [-10, 0, 10], [-3, 0, 3]], dtype=float) / 16
        gy = np.array([[-3, -10, -3], [0, 0, 0], [3, 10, 3]], dtype=float) / 16
    else:
        gx = np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]], dtype=float)

    grad_x = ndi.convolve(img, gx, mode='reflect')
    grad_y = ndi.convolve(img, gy, mode='reflect')
    return np.hypot(grad_x, grad_y), grad_x, grad_y

    """
    1. 腐蚀脑掩码去除颅骨边缘
    2. 叠加原图强度约束：只保留高亮区域的边缘（病灶在 T2 中为高信号）

def postprocess(edge, min_size=50):

    return closed, mask

def iterative_threshold(img, initial, mask=None):
    """只在 mask 区域内做阈值计算"""
    if mask is None:
        mask = np.ones_like(img, dtype=bool)
    pixels = img[mask]
    T = initial
    for _ in range(50):
        bg = pixels[pixels <= T]
        fg = pixels[pixels > T]
        if len(bg) == 0 or len(fg) == 0:
            break
        new_T = (bg.mean() + fg.mean()) / 2
        if abs(new_T - T) < 0.5:
            break
        T = new_T
    return T, (img > T) & mask

def otsu_threshold(img, mask=None):
    """只在 mask 区域内计算 Otsu 阈值"""
    if mask is None:
        mask = np.ones_like(img, dtype=bool)
    pixels = img[mask]
    hist, bins = np.histogram(pixels, bins=256, range=(pixels.min(), pixels.max()))
    bin_centers = (bins[:-1] + bins[1:]) / 2
    weight1 = np.cumsum(hist)
    weight2 = np.cumsum(hist[::-1])[::-1]
    mean1 = np.cumsum(hist * bin_centers) / weight1
    mean2 = (np.cumsum((hist * bin_centers)[::-1]) / weight2[::-1])[::-1]
    variance12 = weight1[:-1] * weight2[1:] * (mean1[:-1] - mean2[1:])**2
    idx = np.argmax(variance12)
    T = bin_centers[idx]
    return T, (img > T) & mask
